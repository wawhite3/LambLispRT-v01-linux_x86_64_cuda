507 LL_float32 coerce_float32()
508 {
509 ME("Cell::coerce_float32()");
510 LL_int32 typ = type();
511 if (typ == T_FLOAT32) return as_float32();
512 else if (typ == T_INT32) return (LL_float32) as_int32();
513 else if (typ == T_INT64) return (LL_float32) as_int64();
514 else if (typ == T_FLOAT64) return (LL_float32) as_float64();
515 else if (typ == T_RATIONAL) {

517 LL_float64 d = (cdr_word_type() == Wt_int32) ? (LL_float64) cdr_int32() : ((Cell*) _cdr_)->coerce_float64();
518 return (LL_float32) (n / d);
519 }

521 else if (typ == T_CHAR) return (LL_float32) as_codepoint();
522 THROW_BAD_TYPE;

525 LL_float64 coerce_float64()
526 {
527 ME("Cell::coerce_float64()");
528 LL_int32 typ = type();
529 if (typ == T_FLOAT64) return as_float64();
530 else if (typ == T_FLOAT32) return (LL_float64) as_float32();
531 else if (typ == T_INT32) return (LL_float64) as_int32();
532 else if (typ == T_INT64) return (LL_float64) as_int64();
533 else if (typ == T_RATIONAL) {
534 LL_float64 n = (car_word_type() == Wt_int32) ? (LL_float64) car_int32() : ((Cell*) _car_)->coerce_float64();
535 LL_float64 d = (cdr_word_type() == Wt_int32) ? (LL_float64) cdr_int32() : ((Cell*) _cdr_)->coerce_float64();
536 return n / d;
537 }
538 else if (typ == T_BIGNUM) {
539 // limb-by-limb: payload = [int32 sign_nlimbs][LL_int32 limb0 ... limbN-1], little-endian.
540 const int32_t *pay = (const int32_t *) as_ByteVec_t();
541 int n = (int)(pay[0] & 0x7fffffff);
542 int neg = (int)((uint32_t) pay[0] >> 31);
543 const LL_int32 *limbs = (const LL_int32 *)(pay + 1);
544 LL_float64 r = 0.0;
545 for (int i = n - 1; i >= 0; i--)
546 r = r * 4294967296.0 + (LL_float64)(uint32_t) limbs[i];
547 return neg ? -r : r;
548 }
549 THROW_BAD_TYPE;
550 }
