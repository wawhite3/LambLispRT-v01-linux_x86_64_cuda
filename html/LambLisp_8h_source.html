107{
108 ll_float_prof[LL_FLOAT_PROF_TOTAL]++;

766 LL_float64 coerce_float64()
767 {
768 ME("Cell::coerce_float64()");
769 LL_int32 typ = type();
770 if (typ == T_FLOAT64) return as_float64();
771 else if (typ == T_FLOAT32) return (LL_float64) as_float32();
772 else if (typ == T_INT32) return (LL_float64) as_int32();
773 else if (typ == T_INT64) return (LL_float64) as_int64();
774 else if (typ == T_RATIONAL) {
775 LL_float64 n = (car_is_int()) ? (LL_float64) car_int32() : ((Cell*) _car_)->coerce_float64();
776 LL_float64 d = (cdr_is_int()) ? (LL_float64) cdr_int32() : ((Cell*) _cdr_)->coerce_float64();
777 return n / d;
778 }
779 else if (typ == T_BIGNUM) {
780 // limb-by-limb: payload = [int32 sign_nlimbs][LL_int32 limb0 ... limbN-1], little-endian.
781 const int32_t *pay = (const int32_t *) as_ByteVec_t();
782 int n = (int)(pay[0] & 0x7fffffff);
783 int neg = (int)((uint32_t) pay[0] >> 31);

2013// load-bearing -- and through apply it merely re-rooted `c`, already reachable from the rooted args.
