764 LL_String concat(const char *s1, const char *s2) {
765 int n1 = strlen(s1);
766 int n2 = strlen(s2);
767 char *s3 = new char[n1 + n2 + 1];
768
769 char *p = s3;
770 while (n1--) *p++ = *s1++;
771 while (n2--) *p++ = *s2++;
772 *p = '\0';
773 return s3;
774 }

776 LL_String &append(const char *s1) {
777 const char *s0 = ptr;
778 int n0 = s0 ? strlen(s0) : 0;
779 int n1 = strlen(s1);
780 char *s2 = new char[n0 + n1 + 1];
781
782 char *p = s2;
783 while (n0--) *p++ = *s0++;
784 while (n1--) *p++ = *s1++;
785 *p = '\0';
786
787 if (ptr) { delete[] ptr; ptr = 0; }
788 ptr = s2;
789 return *this;
790 }

1118inline int ll_u8_seqlen(unsigned char b0)
1119{
1120 if (b0 < 0x80) return 1;
1121 if ((b0 & 0xE0) == 0xC0) return 2;
1122 if ((b0 & 0xF0) == 0xE0) return 3;
1123 if ((b0 & 0xF8) == 0xF0) return 4;
1124 return 1;
1125}
1126
1129inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1130{
1131 unsigned char b0 = (unsigned char) p[0];
1132 int n = ll_u8_seqlen(b0);
1133 if (nb) *nb = n;
1134 if (n == 1) return (ll_codepoint_t) b0;
1135 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1136 for (int i = 1; i < n; i++) {
1137 unsigned char c = (unsigned char) p[i];
1138 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1139 cp = (cp << 6) | (c & 0x3F);
1140 }
1141 return cp;
1142}
1143
1145inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1146{
1147 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1148 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1149 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1150 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1151 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1152 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1153}

1211//The AsciiConverter class mitigates printf formatting problems caused by different int sizes on different processors, when using printf-style format string (%u, %d %lu %ld etc).
1212class AsciiConverter {
1213public:
1214
1215 char *dec(Word_t n) {
1216 LL_int32 ix = buffer_size - 1;
1217 char *ptr = &(buffer[ix]);
1218
1219 *ptr = '\0';
1220 do {
1221 Word_t d = n % 10;
1222 n /= 10;
1223 char c = d + '0';
1224 *--ptr = c;
1225 } while (n);
1226 return ptr;
1227 }
1228
1229 char *dec(LL_int32 n) {
1230 if (n >= 0) return dec((Word_t) n);
1231 if (n == INT32_MIN) return dec((LL_int64) n);
1232 char *ptr = dec(-n);
1233 *--ptr = '-';
1234 return ptr;
1235 }

1251 template <typename T>
1252 typename std::enable_if<std::is_same<T, int>::value && !std::is_same<LL_int32, int>::value,
1253 char *>::type
1254 dec(T n) { return dec((LL_int32) n); }
1255
1256 char *dec(LL_int64 n) {
1257 // Full 64-bit signed decimal; avoids conflict with dec(Word_t) on 64-bit platforms.
1258 LL_int32 ix = buffer_size - 1;
1259 char *ptr = &(buffer[ix]);
1260 *ptr = '\0';
1261 bool neg = (n < 0);
1262 uint64_t uv = neg ? (n == INT64_MIN ? (uint64_t) 9223372036854775808ULL : (uint64_t) -n)
1263 : (uint64_t) n;
1264 do { *--ptr = (char)('0' + (uv % 10)); uv /= 10; } while (uv);
1265 if (neg) *--ptr = '-';
1266 return ptr;
1267 }
1268
1269 char *hex(Word_t n) {
1270 LL_int32 ix = buffer_size - 1;
1271 char *ptr = &(buffer[ix]);
1272 LL_int32 nibs = sizeof(Word_t) * 2;
1273
1274 *ptr = '\0';
1275 while (nibs--) {
1276 Word_t d = n & 0x0f;
1277 n >>= 4;
1278 char c = (d < 10) ? (d + '0') : (d - 10 + 'a');
1279 *--ptr = c;
1280 }
1281 return ptr;
1282
1283 }

1285 char *dec(LL_float32 n) {
1286 snprintf(buffer, buffer_size, "%f", (LL_float64) n);
1287 return buffer;
1288 }
1289
1290private:
1291 static const LL_int32 buffer_size = 128;
1292 char buffer[buffer_size];
1293};
