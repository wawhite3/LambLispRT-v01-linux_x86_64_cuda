1342inline int ll_u8_seqlen(unsigned char b0)
1343{
1344 if (b0 < 0x80) return 1;
1345 if ((b0 & 0xE0) == 0xC0) return 2;
1346 if ((b0 & 0xF0) == 0xE0) return 3;
1347 if ((b0 & 0xF8) == 0xF0) return 4;
1348 return 1;
1349}
1350
1353inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1354{
1355 unsigned char b0 = (unsigned char) p[0];
1356 int n = ll_u8_seqlen(b0);
1357 if (nb) *nb = n;
1358 if (n == 1) return (ll_codepoint_t) b0;
1359 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1360 for (int i = 1; i < n; i++) {
1361 unsigned char c = (unsigned char) p[i];
1362 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1363 cp = (cp << 6) | (c & 0x3F);
1364 }
1365 return cp;
1366}
1367
1369inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1370{
1371 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1372 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1373 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1374 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1375 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1376 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1377}
