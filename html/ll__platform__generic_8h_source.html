698class LL_String {
699public:
700
701 LL_String() { _init(); set(""); }
702 LL_String(const char *other) { _init(); set(other); }
703 LL_String(const char *other, int n) { _init(); set(other, n); }
704 LL_String(const LL_String &other) { _init(); set(other.ptr, other.len); }
705
706 ~LL_String() { if (ptr) { delete[] ptr; ptr = 0; } len = cap = 0; }
707
708 void set(const char *s) { set(s, s ? (int) strlen(s) : 0); }
709
710 void set(const char *s, int n) {
711 if (!s || n <= 0) { _ensure(0); ptr[0] = '\0'; len = 0; return; }
712 if (ptr && s >= ptr && s <= ptr + len) {
713 const int off = (int) (s - ptr);
714 memmove(ptr, ptr + off, (size_t) n);
715 }
716 else {
717 _ensure(n);
718 memmove(ptr, s, (size_t) n);
719 }
720 ptr[n] = '\0';
721 len = n;
722 }
723
724 LL_String concat(const char *s1, const char *s2) {
725 LL_String r;
726 const int n1 = s1 ? (int) strlen(s1) : 0;
727 const int n2 = s2 ? (int) strlen(s2) : 0;
728 r._ensure(n1 + n2);
729 if (n1) memmove(r.ptr, s1, (size_t) n1);
730 if (n2) memmove(r.ptr + n1, s2, (size_t) n2);
731 r.ptr[n1 + n2] = '\0';
732 r.len = n1 + n2;
733 return r;
734 }
735
736 LL_String &append(const char *s1) {
737 if (s1 && *s1) {
738 const int n1 = (int) strlen(s1);
739 const bool self = (ptr && s1 >= ptr && s1 <= ptr + len);
740 const int off = self ? (int) (s1 - ptr) : 0;
741 _ensure(len + n1);
742 memmove(ptr + len, self ? ptr + off : s1, (size_t) n1);
743 len += n1;
744 ptr[len] = '\0';
745 }
746 return *this;
747 }
748
751 LL_String &append(char c) {
752 _ensure(len + 1);
753 ptr[len++] = c;
754 ptr[len] = '\0';
755 return *this;
756 }
757
758 LL_String substring(int start, int len_) { return LL_String(&(ptr[start]), len_); }
759
760 LL_String &operator=(const char *other) { set(other); return *this; }
761 LL_String &operator=(const LL_String &other) { if (this != &other) set(other.ptr, other.len); return *this; }
762
763 bool operator==(const char *other) { return strcmp(ptr, other) == 0; }
764 bool operator!=(const char *other) { return strcmp(ptr, other) != 0; }
765 bool operator==(const LL_String &other) { return strcmp(ptr, other.ptr) == 0; }
766 bool operator!=(const LL_String &other) { return strcmp(ptr, other.ptr) != 0; }
767 char &operator[](int ix) { return ptr[ix]; }
768
769 LL_String operator+(const char *other) { return concat(ptr, other); }
770 LL_String operator+(const LL_String &other) { return concat(ptr, other.ptr); }
771 LL_String operator+(char c) { char cstr[2]; cstr[0] = c; cstr[1] = '\0'; return concat(ptr, cstr); }
772
773 LL_String &operator+=(const char *other) { return append(other); }
774 LL_String &operator+=(const LL_String &other) { return append(other.ptr); }
775 LL_String &operator+=(char c) { return append(c); }
776
777 char *c_str() const { return ptr; }
778 int length() const { return len; }
779
780private:
781 void _init() { ptr = 0; len = 0; cap = 0; }
782
785 void _ensure(int need) {
786 if (cap >= need + 1) return;
787 int ncap = cap ? cap : 16;
788 while (ncap < need + 1) ncap += ncap;
790 char *np = new char[ncap];
791 if (ptr) { memmove(np, ptr, (size_t) len + 1); delete[] ptr; }
792 else { np[0] = '\0'; }
793 ptr = np;
794 cap = ncap;
795 }
796
797 char *ptr;
798 int len;
799 int cap;
800};

1181inline int ll_u8_seqlen(unsigned char b0)
1182{
1183 if (b0 < 0x80) return 1;
1184 if ((b0 & 0xE0) == 0xC0) return 2;
1185 if ((b0 & 0xF0) == 0xE0) return 3;
1186 if ((b0 & 0xF8) == 0xF0) return 4;
1187 return 1;
1188}
1189
1192inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1193{
1194 unsigned char b0 = (unsigned char) p[0];
1195 int n = ll_u8_seqlen(b0);
1196 if (nb) *nb = n;
1197 if (n == 1) return (ll_codepoint_t) b0;
1198 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1199 for (int i = 1; i < n; i++) {
1200 unsigned char c = (unsigned char) p[i];
1201 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1202 cp = (cp << 6) | (c & 0x3F);
1203 }
1204 return cp;
1205}
1206
1208inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1209{
1210 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1211 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1212 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1213 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1214 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1215 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1216}

1274//The AsciiConverter class mitigates printf formatting problems caused by different int sizes on different processors, when using printf-style format string (%u, %d %lu %ld etc).
1275class AsciiConverter {
1276public:
1277
1278 char *dec(Word_t n) {
1279 char *ptr = slot_end();
1280
1281 *ptr = '\0';
1282 do {
1283 Word_t d = n % 10;
1284 n /= 10;
1285 char c = d + '0';
1286 *--ptr = c;
1287 } while (n);
1288 return ptr;
1289 }
1290
1291 char *dec(LL_int32 n) {
1292 if (n >= 0) return dec((Word_t) n);
1293 if (n == INT32_MIN) return dec((LL_int64) n);
1294 char *ptr = dec(-n);
1295 *--ptr = '-';
1296 return ptr;

2097 * gap: the rootstack is unwound at the MOP3 BOUNDARY, by `mop3_catch`, which restores the depth
2098 * `mop3_try` saved on entry. An `ll_catch` frame between the throw and that boundary has nothing

2103 * `mop3_catch` APPLICATION error -- restore to the mop3 boundary so the REPL continues.
